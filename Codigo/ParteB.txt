start:
    .include "m2560def.inc"

	; Le cambiamos nombre a los registros 16 y 17 para entradas y salidas
	.def entradas = r16
	.def salidas = r17

	.cseg
	.org 0x00
 

	; PB0, PB1 y PB2 como entradas
	ldi entradas, 0b00000000
	out DDRB, entradas

	; Pull-up internas en PB0, PB1 y PB2
	ldi entradas, 0b00000111
	out PORTB, entradas

	; PD0-PD3 como salidas
	ldi salidas, 0b00001111
	out DDRD, salidas

	; PC0-PC2 como salidas
	ldi salidas, 0b00000111
	out DDRC, salidas

	main_loop:

        loop_cero:
        sbrc r18, 0  
		rcall confirmacion

		; Cero
		sbi PORTD, PD0
		sbi PORTD, PD1
		sbi PORTD, PD2
		sbi PORTD, PD3
		sbi PORTC, PC0
		sbi PORTC, PC1
		cbi PORTC, PC2
		
        ; 312.5ns
        sbis PINB, PB1
		ori  r18, 0b00000001
		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_uno

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_quince

		rjmp loop_cero

		loop_uno:
        sbrc r18, 0  
		rcall confirmacion

		; Uno
		cbi PORTD, PD0
		sbi PORTD, PD1
		sbi PORTD, PD2
		cbi PORTD, PD3
		cbi PORTC, PC0
		cbi PORTC, PC1
		cbi PORTC, PC2

        sbis PINB, PB1
		ori  r18, 0b00000001
		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_dos

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_cero

		rjmp loop_uno

		loop_dos:
        sbrc r18, 0  
		rcall confirmacion

		; Dos
		sbi PORTD, PD0
		sbi PORTD, PD1
		cbi PORTD, PD2
		sbi PORTD, PD3
		sbi PORTC, PC0
		cbi PORTC, PC1
		sbi PORTC, PC2

        sbis PINB, PB1
		ori  r18, 0b00000001
		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_tres

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_uno

		rjmp loop_dos

		loop_tres:
        sbrc r18, 0  
		rcall confirmacion

		; Tres
		sbi PORTD, PD0
		sbi PORTD, PD1
		sbi PORTD, PD2
		sbi PORTD, PD3
		cbi PORTC, PC0
		cbi PORTC, PC1
		sbi PORTC, PC2

        sbis PINB, PB1
		ori  r18, 0b00000001
		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_cuatro

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_dos

		rjmp loop_tres

		loop_cuatro:
        sbrc r18, 0  
		rcall confirmacion

		; Cuatro
		cbi PORTD, PD0
		sbi PORTD, PD1
		sbi PORTD, PD2
		cbi PORTD, PD3
		cbi PORTC, PC0
		sbi PORTC, PC1
		sbi PORTC, PC2

        sbis PINB, PB1
		ori  r18, 0b00000001
		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_cinco

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_tres

		rjmp loop_cuatro

		loop_cinco:
        sbrc r18, 0  
		rcall confirmacion

		; Cinco
		sbi PORTD, PD0
		cbi PORTD, PD1
		sbi PORTD, PD2
		sbi PORTD, PD3
		cbi PORTC, PC0
		sbi PORTC, PC1
		sbi PORTC, PC2

        sbis PINB, PB1
		ori  r18, 0b00000001
		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_seis

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_cuatro

		rjmp loop_cinco

		loop_seis:
        sbrc r18, 0  
		rcall confirmacion

		; Seis
		sbi PORTD, PD0
		cbi PORTD, PD1
		sbi PORTD, PD2
		sbi PORTD, PD3
		sbi PORTC, PC0
		sbi PORTC, PC1
		sbi PORTC, PC2

        sbis PINB, PB1
		ori  r18, 0b00000001
		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_siete

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_cinco

		rjmp loop_seis

		loop_siete:
        sbrc r18, 0  
		rcall confirmacion

		; Siete
		sbi PORTD, PD0
		sbi PORTD, PD1
		sbi PORTD, PD2
		cbi PORTD, PD3
		cbi PORTC, PC0
		cbi PORTC, PC1
		cbi PORTC, PC2

        sbis PINB, PB1
		ori  r18, 0b00000001
		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_ocho

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_seis

		rjmp loop_siete

		loop_ocho:
        sbrc r18, 0  
		rcall confirmacion

		; Ocho
		sbi PORTD, PD0
		sbi PORTD, PD1
		sbi PORTD, PD2
		sbi PORTD, PD3
		sbi PORTC, PC0
		sbi PORTC, PC1
		sbi PORTC, PC2

        sbis PINB, PB1
		ori  r18, 0b00000001
		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_nueve

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_siete

        sbrc r18, 0

		; Nueve

		cbi PORTC, PC0

        sbis PINB, PB1

		rjmp loop_cero
		
        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_diez

		ori  r18, 0b00000001

		; Diez

		sbi PORTC, PC0

        sbis PINB, PB1

		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_once

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_nueve

		rjmp loop_diez

		loop_once:
        sbrc r18, 0  
		rcall confirmacion

		; Once
		cbi PORTD, PD0
		cbi PORTD, PD1
		sbi PORTD, PD2
		sbi PORTD, PD3
		sbi PORTC, PC0
		sbi PORTC, PC1
		sbi PORTC, PC2

        sbis PINB, PB1
		ori  r18, 0b00000001
		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_doce

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_nueve

		rjmp loop_once

		loop_doce:
        sbrc r18, 0  
		rcall confirmacion

		; Doce
		sbi PORTD, PD0
		cbi PORTD, PD1
		cbi PORTD, PD2
		sbi PORTD, PD3
		sbi PORTC, PC0
		sbi PORTC, PC1
		cbi PORTC, PC2

        sbis PINB, PB1
		ori  r18, 0b00000001
		sbis PINB, PB1

        sbis PINB, PB0

		rjmp loop_trece

		sbis PINB, PB2
		ori  r18, 0b00000001

		rjmp loop_diez

        sbis PINB, PB0

		rjmp loop_doce

		rjmp loop_trece

		sbi PORTD, PD3

		ori  r18, 0b00000001
		sbis PINB, PB1
		rjmp loop_cero

        sbis PINB, PB0
		ori  r18, 0b00000001
		sbis PINB, PB0
		rjmp loop_cero

		sbis PINB, PB2
		ori  r18, 0b00000001
		sbis PINB, PB2
		rjmp loop_catorce

		rjmp loop_quince

    rjmp main_loop 

    	; 22.6 ms
	delay:
		ldi r20, 3
		even_outer_db:
			ldi r21, 200
		outer_db:
			ldi r22, 200
		inner_db:
			dec r22
			brne inner_db
			dec r21
			brne outer_db
			dec r20
			brne even_outer_db
			ret


	debounce:
		rcall delay

		sbis PINB, PB0
		rjmp debounce

		sbis PINB, PB1
		rjmp debounce

		sbis PINB, PB2
		rjmp debounce

		ret


	confirmacion:
		rcall delay

		rcall debounce

		andi r18, 0b11111110

		ret
