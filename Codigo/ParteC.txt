.cseg
.org 0x00

main:
; Configuro Port A para LEDs
    ldi r16, 0xFF ; carga unos

; Configuro Port B para Botones
	ldi r16,0x00 ; carga 0's

    ev_limites:
    cpi r17, 9 ; después de pasar la secuancia 8 
    breq ir_pri; vuelve a la primera secuencia
    cpi r17, 0; después de pasar por la primera secuencia
    breq ir_ul; vuelve a la octava secuencia
    rjmp ev_Sec

ir_pri:

    rjmp ev_Sec

ir_ul:
    ldi r17, 8
    rjmp ev_Sec
